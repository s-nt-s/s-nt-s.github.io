## Instalar NickelMenu

Conectar Kobo por usb.

```
$ wget -q -O "/media/$USER/KOBOeReader/.kobo/KoboRoot.tgz" https://github.com/pgaskin/NickelMenu/releases/latest/download/KoboRoot.tgz
$ mkdir -p "/media/$USER/KOBOeReader/.adds/" "/media/$USER/KOBOeReader/.adds/nm/"
$ cat > /media/$USER/KOBOeReader/.adds/nm/config <<'EOF'
experimental:menu_main_15505_label:Atajos

menu_item:main:Instapaper:nickel_open:library:instapaper
menu_item:main:KOReader:cmd_spawn:quiet:exec /mnt/onboard/.adds/koreader/koreader.sh
menu_item:main:Refrescar Biblioteca:nickel_misc:rescan_books_full
menu_item:main:Modo USB:nickel_misc:force_usb_connection

menu_item :main : SSH (toggle) :cmd_output :1000:sh /mnt/onboard/.adds/koreader/scripts/mydropbear.sh 22

menu_item:main:Reiniciar:nickel_misc:home
  chain_success:power:reboot

menu_item:selection_search:Buscar en DuckDuckGo :nickel_browser :modal:https://duckduckgo.com/?q={1|S|%}

menu_item:reader:Modo oscuro:nickel_setting:toggle:dark_mode
menu_item:reader:Apagar WiFi:nickel_wifi:disable
EOF
```

Reiniciar

## Instalar KoReader

```
$ wget -qO- https://github.com/koreader/koreader/releases/download/v2026.07.1/koreader-kobo-v2026.07.1.zip | bsdtar -xf - -C "/media/$USER/KOBOeReader/.adds/"
$ mkdir -p "/media/$USER/KOBOeReader/.adds/koreader/scripts/"
$ nano "/media/$USER/KOBOeReader/.adds/koreader/scripts/mydropbear.sh"
```

```python
#!/bin/sh
TMP_PID="/tmp/dropbear_koreader.pid"
PORT="$1"

if [ -f "$TMP_PID" ]; then
   cat "$TMP_PID" | xargs kill
   rm -f "$TMP_ID"
   echo "SSH parado"
   exit 0
fi
IP=$(/sbin/ifconfig | /usr/bin/awk '/inet addr/{print substr($2,6)}')

if [ -z "$IP" ]; then
   IP="127.0.0.1"
fi

/bin/mount -t devpts | /bin/grep -q /dev/pts || { /bin/mkdir -p /dev/pts && /bin/mount -t devpts devpts /dev/pts; }
if [ ! -d /dev/pts ]; then
   echo "Error al crear /dev/pts"
   exit 1
fi

export PATH=$PATH:/mnt/onboard/.adds/koreader/scripts
export LD_LIBRARY_PATH=$LD_LIBRARY_PATH:/mnt/onboard/.adds/koreader/libs
export HOME=/mnt/onboard/.adds/koreader/
cd $HOME
./dropbear -E -R -p "$PORT" -P "$TMP_PID"

if [ $? -ne 0 ]; then
   echo "Error al inciar SSH en $IP:$PORT"
   exit 1
fi

echo "SSH $IP:$PORT iniciado"
exit 0
```

## Eliminar tienda kobo y otras configuraciones

Editar `"/media/$USER/KOBOeReader/.kobo/Kobo/Kobo eReader.conf"` y añadir:

```
[ApplicationPreferences]
SideloadedMode=true
AIRPLANE_MODE=true

[BluetoothSettings]
Alias=eLibro

[FeatureSettings]
MyWords=true
ExcludeSyncFolders=(\\.(?!kobo|adobe).+|([^.][^/]*/)+\\..+)

[ReadingLife]
LastActivityNavItem=0

[PowerOptions]
AutoColorEnabled=true
AutoOffMinutes=30
AutoSleepMinutes=15
FrontLightLevel=0
IndicatorCharging=false
RecentlyViewedEnabled=false
ShowBookCover=false

[Reading]
numPartialUpdatePageTurns=10
readingAdobeShowPageNumbers=false
showProgressBar=false
swipeBrightness=false
```
